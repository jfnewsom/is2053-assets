# ============================================================
# CITY DATA LOADERS (Added in Lab 4-1)
# ============================================================

def load_city_distances(filename):
    """
    Read the city routes from a file into a nested dictionary.

    Args:
        filename (str): The file to read, one city1,city2,miles per line

    Returns:
        dict: distances[city1][city2] is the miles from city1 to city2
    """
    distances = {}
    infile = open(filename, 'r')
    for line in infile:
        parts = line.strip().split(',')
        city1 = parts[0]
        city2 = parts[1]
        miles = int(parts[2])
        if city1 not in distances:
            distances[city1] = {}
        distances[city1][city2] = miles
    infile.close()
    return distances

def load_city_info(filename):
    """
    Read each city's type and tagline from a file into a dictionary.

    Args:
        filename (str): The file to read, one city,type,tagline per line

    Returns:
        dict: info[city] is {'type': ..., 'tagline': ...}
    """
    info = {}
    infile = open(filename, 'r')
    for line in infile:
        parts = line.strip().split(',')
        city = parts[0]
        city_type = parts[1]
        tagline = parts[2]
        info[city] = {'type': city_type, 'tagline': tagline}
    infile.close()
    return info

# ============================================================
# DESTINATION HELPERS (Added in Lab 3-2, rewritten in Lab 4-1)
# ============================================================

def display_destinations(city, distances):
    """
    Show the numbered destination menu for a city.

    Args:
        city (str): The city the player is in
        distances (dict): The nested dictionary from load_city_distances()

    Returns:
        int: How many destinations the menu shows
    """
    print(f'From {city}, you can travel to:')
    option_num = 1
    for dest in distances[city]:
        miles = distances[city][dest]
        print(f'  {option_num}. {dest} - {miles} miles')
        option_num = option_num + 1
    return len(distances[city])


def get_destination(city, distances, choice):
    """
    Look up the destination for a menu choice.

    Args:
        city (str): The city the player is in
        distances (dict): The nested dictionary from load_city_distances()
        choice (int): The menu number the player chose, from 1

    Returns:
        tuple: The destination and its distance
    """
    dests = list(distances[city].keys())
    dest = dests[choice - 1]
    return dest, distances[city][dest]


def get_redirect(city, distances, avoid):
    """
    Pick a random destination other than the one the player wanted.

    Args:
        city (str): The city the player is in
        distances (dict): The nested dictionary from load_city_distances()
        avoid (str): The destination the player wanted

    Returns:
        tuple: Another destination and its distance, or the only one
        there is when the city has a single route
    """
    options = list(distances[city].keys())
    if avoid in options:
        options.remove(avoid)
    if len(options) > 0:
        index = random.randint(0, len(options) - 1)
        dest = options[index]
        return dest, distances[city][dest]
    else:
        only_option = list(distances[city].keys())[0]
        return only_option, distances[city][only_option]
