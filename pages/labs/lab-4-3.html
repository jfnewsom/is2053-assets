# ============================================================
# TRIVIA FUNCTIONS (Added in Lab 4-3)
# ============================================================

def load_trivia(filename):
    """
    Read the trivia from a file into a dictionary of lists.

    Args:
        filename (str): The file to read, one city,question,answer per line

    Returns:
        dict: trivia[city] is a list of {'question': ..., 'answer': ...}
    """
    trivia = {}
    infile = open(filename, 'r')
    for line in infile:
        parts = line.strip().split(',')
        city = parts[0]
        question = parts[1]
        answer = parts[2]
        if city not in trivia:
            trivia[city] = []
        trivia[city].append({'question': question, 'answer': answer})
    infile.close()
    return trivia

# ============================================================

def clean_input(prompt):
    """
    Ask a question and return the answer without spaces, in capitals.

    Args:
        prompt (str): The prompt to show

    Returns:
        str: What the player typed, stripped and uppercased
    """
    return input(prompt).strip().upper()

def trivia_gate(dest, trivia, current, distances):
    """
    Ask a TRUE or FALSE question about the destination. A right answer
    earns the trip; a wrong one sends the player somewhere else.

    Args:
        dest (str): The destination the player chose
        trivia (dict): The dictionary from load_trivia()
        current (str): The city the player is in
        distances (dict): The nested dictionary from load_city_distances()

    Returns:
        tuple: Where the player goes, its distance, and True if the
        answer was right
    """
    print()
    print('--- TRIVIA TIME! ---')
    print(f'To reach {dest}, answer this question:')
    print()
    q = random.choice(trivia[dest])
    print(f'  {q["question"]}')
    print()
    answer = function_library.clean_input('TRUE or FALSE? ')
    if answer == 'T' or answer == 'TRUE':
        player_answer = 'TRUE'
    elif answer == 'F' or answer == 'FALSE':
        player_answer = 'FALSE'
    else:
        print('Invalid answer! Counting as wrong.')
        player_answer = 'INVALID'
    if player_answer == q['answer']:
        print('CORRECT! You earned your destination.')
        return dest, distances[current][dest], True
    else:
        print(f'WRONG. The answer was {q["answer"]}.')
        redirect, redirect_miles = function_library.get_redirect(
            current, distances, dest)
        print(f'Detour! You ended up going to {redirect} instead.')
        return redirect, redirect_miles, False
